// ---------
// http spec
// ---------
GET /                                 // presents login/signup page (login.html)
GET /login?username=str&password=str  // 200 response body will contain a token
GET /signup?username=str&password=str // 200 response body will contain a token
GET /app?token=str                    // presents main app (app.html) - token will be sent over websocket to initialize connection
GET /file?id=int&token=str            // 200 response body contains file
PUT /file?name=str&token=str          // request body contains file, 200 response body contains file id

// --------------
// websocket spec
// --------------

// client -> server
{ type: "token", token: str }
{ type: "getGroupInfo", id: int }
{ type: "getMessages", channelId: int, index: int } // index -1 for latest
{ type: "typingStatus", isTyping: bool }
{ type: "sendMessage", channelId: int, contents: str, fileId: int? }
{ type: "getUserInfo", username: str }
{ type: "setPfp", fileId: int }
{ type: "createThing", thingType: str, name: str }
{ type: "renameThing", thingType: str, id: int, name: str }
{ type: "deleteThing", thingType: str, id: int }
{ type: "inviteUser", groupId: int, userId: int }

// server -> client
{ type: "invalidToken" }
{ type: "groupList", groups: [ { id: int, name: str } ] }
{ type: "groupInfo", id: int, channels: [ { id: int, name: str } ] }
{ type: "messages", channelId: int, messages: [ { index: int, userId: int, contents: str, fileId: int? } ] }
{ type: "typingIndicator", channelId: int, usersTyping: [ int ] }
{ type: "userInfo", id: int, name: str, pfpId: int }
